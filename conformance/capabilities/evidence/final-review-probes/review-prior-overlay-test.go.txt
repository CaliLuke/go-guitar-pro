func TestFreshPriorLegacyAutomationEdits(t *testing.T){

 s.PanAutomations=[]PanAutomation{{Position:.25,Value:.125,Linear:true},{Position:.25,Value:.875},{Position:.75,Value:.625}}
 s.Tracks[0].Sounds=[]TrackSound{{Name:"One & α",Path:"review/one",Role:"main",Program:27},{Name:"Two",Path:"review/two",Role:"solo",Program:27}}
 s.Tracks[0].SoundAutomations=[]SoundAutomation{{Position:0,Sound:1,Text:"one"},{Position:.25,Sound:0,Text:"two",Linear:true},{Position:.25,Sound:1,Text:"three"}}
 for _,clear:=range []bool{false,true}{
 if clear{s.PanAutomations=nil;s.Tracks[0].SoundAutomations=nil}
 before,_:=json.Marshal(s); data,report,err:=ExportWithReport(s,ExportFormatGP8,ExportOptions{});if err!=nil{t.Fatal(err,report)}
 out,err:=Parse(data);if err!=nil{t.Fatal(err)}
 if !slices.Equal(out.PanAutomations,s.PanAutomations)||!slices.Equal(out.Tracks[0].Sounds,s.Tracks[0].Sounds)||!slices.Equal(out.Tracks[0].SoundAutomations,s.Tracks[0].SoundAutomations){t.Fatalf("clear=%t stale raw events resurrected",clear)}
 after,_:=json.Marshal(s);if string(before)!=string(after){t.Fatal("source mutated")}
 var facts struct{Sound []conformanceSoundEvent};readAlphaTabOracleFacts(t,"--automations",writeConformanceFixture(t,data),&facts)
 if clear{if len(facts.Sound)!=1||facts.Sound[0].Value!=27||facts.Sound[0].Position!=0||facts.Sound[0].Text!=""{t.Fatal("clearing must leave only consumer default program",facts)};continue};if len(facts.Sound)!=len(s.Tracks[0].SoundAutomations){t.Fatal(facts)}
 for i,a:=range facts.Sound{want:=s.Tracks[0].SoundAutomations[i];if a.Value!=27||a.Position!=want.Position||a.Text!=want.Text||a.Linear!=want.Linear{t.Fatal(a,want)}}
 }
}
