 records=append(records,binaryStyleRecord{key:"Review/opaqueColor",kind:7,value:[]byte{12,34,56,78}})
 records=append(records,binaryStyleRecord{key:"System/ExtendedBarLines",kind:0,value:[]byte{1}},scoreDisplayInteger("System/barIndexDrawType",2))
 records=append(records,headerFooterRecords(s)...)
 for _,k:=range freshDisplayKeys{records=append(records,binaryStyleRecord{key:k,kind:0,value:[]byte{1}})}
 for _,k:=range []string{"System/bracketExtendMode","System/trackNameModeSingle","System/trackNameModeMulti"}{records=append(records,scoreDisplayInteger(k,2))}
 source:=conformanceBackingArchive(t,xml,map[string][]byte{"Content/BinaryStylesheet":encodeDisplayTestRecords(records)})
 parsed,e:=Parse(source);if e!=nil{t.Fatal(e)}
 for _,f:=range freshDisplayFields{p:=reflect.ValueOf(parsed.Style).Elem().FieldByName(f);if p.IsNil()||!p.Elem().Bool(){t.Fatal("last duplicate did not win",f)}}
 if *parsed.Style.Brackets!=2||*parsed.Style.SingleTrackNameMode!=2||*parsed.Style.MultiTrackNameMode!=2{t.Fatal("integer duplicate lost")}
 // PartConfiguration is a separate archive member; restore only the authored score rest preference for this fixture.
 parsed.Style.MultiRest=ptrTo(true)
 before,_:=json.Marshal(parsed);out,e:=Export(parsed,ExportFormatGP8);if e!=nil{t.Fatal(e)}
 after,_:=json.Marshal(parsed);if string(before)!=string(after){t.Fatal("export mutation")}
 var input,output scoreStyleFacts;readAlphaTabOracleFacts(t,"--score-style",writeConformanceFixture(t,source),&input);readAlphaTabOracleFacts(t,"--score-style",writeConformanceFixture(t,out),&output)
 if !reflect.DeepEqual(input,output){t.Fatal("duplicate collapse changed raw consumer",input,output)}
 all:=append(slices.Clone(freshDisplayKeys),"System/bracketExtendMode","System/trackNameModeSingle","System/trackNameModeMulti")
 wire:=scoreStyleWire(t,out);for _,k:=range all{n:=0;for _,r:=range wire{if r.key==k{n++}};if n!=1{t.Fatal(k,n)}}

 out,e=Export(parsed,ExportFormatGP8);if e!=nil{t.Fatal(e)};wire=scoreStyleWire(t,out)
 for _,r:=range wire{if slices.Contains(all,r.key){t.Fatal("cleared key replayed",r)}}
 opaque:=findStyleRecord(t,wire,"Review/opaqueColor");if opaque.kind!=7||!slices.Equal(opaque.value,[]byte{12,34,56,78}){t.Fatal("unknown record changed")}
 readAlphaTabOracleFacts(t,"--score-style",writeConformanceFixture(t,out),&output)
 if !reflect.DeepEqual(output.Headers,input.Headers)||!output.Extended||output.Numbers!="Hide"{t.Fatal("shared header/barlines changed",output)}
 expected:=map[string]any{"hideDynamics":false,"useSystemSignSeparator":false,"globalDisplayTuning":true,"globalDisplayChordDiagramsOnTop":true,"globalDisplayChordDiagramsInScore":false,"singleTrackTrackNamePolicy":float64(1),"multiTrackTrackNamePolicy":float64(1),"firstSystemTrackNameMode":float64(1),"otherSystemsTrackNameMode":float64(1),"firstSystemTrackNameOrientation":float64(1),"otherSystemsTrackNameOrientation":float64(1),"bracketExtendMode":float64(1)}

 var rest multiRestFacts;readAlphaTabOracleFacts(t,"--multi-rest",writeConformanceFixture(t,out),&rest);if !rest.Global{t.Fatal("clearing display lost multi-rest")}
}
